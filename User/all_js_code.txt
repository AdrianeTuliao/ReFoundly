// 1. Setup Dropdown Search Logic
function setupSearch(inputId, listId) {
    const input = document.getElementById(inputId);
    const list = document.getElementById(listId);
    if (!input || !list) return;

    const items = list.querySelectorAll('li');

    input.addEventListener('keyup', function() {
        let filter = this.value.toLowerCase();
        list.style.display = 'block'; 
        items.forEach(item => {
            let text = item.textContent.toLowerCase();
            item.style.display = text.includes(filter) ? "block" : "none";
        });
    });

    items.forEach(item => {
        item.addEventListener('click', function() {
            input.value = this.textContent;
            list.style.display = 'none'; 
        });
    });

    document.addEventListener('click', (e) => {
        if (!input.contains(e.target) && !list.contains(e.target)) list.style.display = 'none';
    });
    
    input.addEventListener('focus', () => list.style.display = 'block');
}

async function updateNotificationsUI() {
    const list = document.getElementById("notifList");
    const badge = document.getElementById("notifBadge");
    if (!list) return;

    try {
        const res = await fetch('/api/user/notifications');

        // 1. Check if the HTTP response is OK (prevents JSON syntax errors on HTML 404/500 responses)
        if (!res.ok) {
            console.warn(`Notifications fetch failed with status: ${res.status}`);
            return;
        }

        const data = await res.json();

        // 2. Validate that data is an array before processing
        if (!Array.isArray(data)) {
            console.error("Notifications response was not an array:", data);
            return;
        }

        // Update Badge Count (Unread only)
        const unreadCount = data.filter(n => !n.is_read).length;
        if (badge) {
            badge.innerText = unreadCount;
            // Use 'flex' so the centering CSS works, 'none' to hide if 0
            badge.style.display = unreadCount > 0 ? "flex" : "none";
        }

        // Filter list based on active tab
        let displayList = currentNotifTab === 'unread' ? data.filter(n => !n.is_read) : data;

        if (displayList.length === 0) {
            list.innerHTML = `<div style="padding: 40px 20px; text-align: center; color: #999; font-size: 0.85rem;">
                No ${currentNotifTab === 'unread' ? 'unread' : ''} notifications
            </div>`;
            return;
        }

        list.innerHTML = displayList.map(n => {
            const date = new Date(n.created_at).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
            
            // 3. Escape the message content to prevent potential XSS issues
            const safeMessage = (n.message || "Notification updated")
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");

            return `
                <div class="notif-item-modern ${n.is_read ? '' : 'unread-bg'}" onclick="handleNotifClick(${n.id}, ${n.item_id || 'null'})">
                    <div class="notif-icon-circle" style="background: #7aa340; width:35px; height:35px; border-radius:50%; color:white; display:flex; align-items:center; justify-content:center; flex-shrink:0;">
                        <i class="fa-solid fa-bell" style="font-size:0.8rem;"></i>
                    </div>
                    <div style="flex-grow:1;">
                        <div style="font-size:0.85rem; color:#1c1e21; font-weight:${n.is_read ? '600' : '700'}">${safeMessage}</div>
                        <div style="font-size:0.75rem; color:#65676b;">${date}</div>
                    </div>
                    ${!n.is_read ? '<div class="unread-dot-small"></div>' : ''}
                </div>
            `;
        }).join('');
    } catch (err) {
        console.error("Notification Load Error:", err);
    }
}

function switchNotifTab(type) {
    currentNotifTab = type;
    document.getElementById('tabAll')?.classList.toggle('active', type === 'all');
    document.getElementById('tabUnread')?.classList.toggle('active', type === 'unread');
    updateNotificationsUI();
}

async function handleNotifClick(id, itemId) {
    try {
        await fetch(`/api/user/notifications/read/${id}`, { method: 'POST' });
    } catch (err) {
        console.error("Failed to mark notification as read:", err);
    }
    if (!itemId) {
        window.location.href = 'history.html';
        return;
    }
    try {
        const res = await fetch(`/api/items/${itemId}`);
        if (res.ok) {
            const item = await res.json();
            const dbType = (item.report_type || "").toLowerCase().trim();
            finalRedirect(itemId, dbType);
        } else {
            window.location.href = 'history.html';
        }
    } catch (err) {
        console.error("Could not resolve matched item:", err);
        window.location.href = 'history.html';
    }
}

window.finalRedirect = function(id, type) {
    const verifiedType = String(type || "").toLowerCase().trim();
    if (verifiedType === "lost") {
        window.location.href = "lost.html?open=" + id;
    } else {
        window.location.href = "found.html?open=" + id;
    }
};

async function loadConversations(options = {}) {
    const silent = options.silent === true;

    if (conversationPollRunning) {
        return conversations;
    }

    conversationPollRunning = true;

    try {
        const response = await fetch(
            "/api/conversations",
            {
                method: "GET",
                headers: {
                    Accept: "application/json"
                },
                cache: "no-store"
            }
        );

        if (response.status === 401) {
            window.location.href = "index.html";
            return [];
        }

        const data = await response.json();

        if (!response.ok || !data.success) {
            throw new Error(
                data.message ||
                "Unable to load conversations."
            );
        }

        const newList =
            Array.isArray(data.conversations)
                ? data.conversations
                : [];

        const changed =
            conversationsChanged(
                conversations,
                newList
            );

        conversations = newList;

        if (!silent || changed) {
            renderConversationList(
                conversations
            );
        }

        return conversations;

    } catch (error) {
        console.error(
            "loadConversations:",
            error
        );

        if (!silent && chatList) {
            chatList.innerHTML = `
                <div class="empty-chat-list">
                    <i class="fas fa-exclamation-circle"></i>
                    <p>Unable to load conversations.</p>
                </div>
            `;
        }

        return [];
    } finally {
        conversationPollRunning = false;
    }
}


function conversationsChanged(oldList, newList) {
    if (oldList.length !== newList.length) {
        return true;
    }

    for (let i = 0; i < newList.length; i++) {
        const oldItem = oldList[i];
        const newItem = newList[i];

        if (
            String(oldItem?.conversation_id) !==
            String(newItem?.conversation_id)
        ) {
            return true;
        }

        if (
            String(oldItem?.last_message || "") !==
            String(newItem?.last_message || "")
        ) {
            return true;
        }

        if (
            Number(oldItem?.unread_count || 0) !==
            Number(newItem?.unread_count || 0)
        ) {
            return true;
        }
    }

    return false;
}

    updateConversationHeader();

    if (chatFooter) {
        chatFooter.style.display = "none";
    }

    if (chatBody) {
        chatBody.innerHTML = `
            <div class="conversation-loading">
                <i class="fas fa-spinner fa-spin"></i>
                <p>Loading conversation...</p>
            </div>
        `;
    }

    try {
        const response = await fetch(
            `/api/conversations/${encodeURIComponent(
                currentConversationId
            )}`,
            {
                method: "GET",
                headers: {
                    Accept:
                        "application/json"
                },
                cache: "no-store"
            }
        );

        if (response.status === 401) {
            window.location.href =
                "index.html";
            return;
        }

        const data =
            await response.json();

        if (!response.ok || !data.success) {
            throw new Error(
                data.message ||
                "Unable to open conversation."
            );
        }

        currentConversation =
            data.conversation ||
            data.data ||
            {};

        updateConversationHeader();

        if (chatFooter) {
            chatFooter.style.display =
                "flex";
        }

        renderAgreementBar();

        await loadMessages(
            currentConversationId,
            true
        );

        await loadAgreementStatus();

        renderConversationList(
            conversations
        );

        startMessagePolling();
        startAgreementPolling();

    } catch (error) {
        console.error(
            "openConversation:",
            error
        );

        if (chatBody) {
            chatBody.innerHTML = `
                <div class="empty-chat-list">
                    <i class="fas fa-exclamation-circle"></i>
                    <p>Unable to open conversation.</p>
                </div>
            `;
        }
    }
}


function updateConversationHeader() {
    if (!currentConversation) {
        if (currentUserName) {
            currentUserName.textContent =
                "Select a conversation";
        }

        if (currentItemName) {
            currentItemName.textContent = "";
        }

        renderItemFloatingCard();
        return;
    }

    if (currentUserName) {
        currentUserName.textContent =
            currentConversation.other_user_name ||
            currentConversation.other_user_email ||
            "User";
    }

    if (currentItemName) {
        currentItemName.textContent =
            currentConversation.item_name
                ? `Item: ${currentConversation.item_name}`
                : "";
    }

    renderItemFloatingCard();

/* =========================================================
   MESSAGES
   ========================================================= */

async function loadMessages(
    conversationId,
    initial = false
) {
    if (!conversationId) return [];

    if (messagePollRunning) {
        return currentMessages;
    }

    messagePollRunning = true;

    try {
        const response = await fetch(
            `/api/conversations/${encodeURIComponent(
                conversationId
            )}/messages`,
            {
                method: "GET",
                headers: {
                    Accept:
                        "application/json"
                },
                cache: "no-store"
            }
        );

        if (response.status === 401) {
            window.location.href =
                "index.html";
            return [];
        }

        const data =
            await response.json();

        if (!response.ok || !data.success) {
            throw new Error(
                data.message ||
                "Unable to load messages."
            );
        }

        const messages =
            Array.isArray(data.messages)
                ? data.messages
                : [];

        const signature =
            makeMessageSignature(
                messages
            );

        if (
            initial ||
            signature !== lastMessageSignature
        ) {
            renderMessages(
                messages,
                !initial
            );

            lastMessageSignature =
                signature;

            currentMessages =
                messages;
        }

        return messages;

    } catch (error) {
        console.error(
            "loadMessages:",
            error
        );

        if (initial && chatBody) {
            chatBody.innerHTML = `
                <div class="empty-chat-list">
                    <i class="fas fa-exclamation-circle"></i>
                    <p>Unable to load messages.</p>
                </div>
            `;
        }

        return [];
    } finally {
        messagePollRunning = false;
    }
}


function makeMessageSignature(
    messages
) {
    if (!messages.length) {
        return "empty";
    }

    return messages
        .map(message => {
            return [
                message.id ||
                    message.message_id ||
                    "",
                message.sender_id ||
                    "",
                message.message ||
                    "",
                message.message_type ||
                    "",
                message.image_url ||
                    "",
                message.created_at ||
                    ""
            ].join("|");
        })
        .join("||");
}


function renderMessages(
    messages,
    preserveScroll = true
) {
    if (!chatBody) return;

    const wasNearBottom =
        isChatNearBottom();

    const previousHeight =
        chatBody.scrollHeight;

    const previousTop =
        chatBody.scrollTop;

    const agreementBar =
        document.getElementById(
            "agreementBar"
        );

    const readyPanel =
        document.getElementById(
            "transactionReadyPanel"
        );

    chatBody.innerHTML = "";

    if (agreementBar) {
        chatBody.appendChild(
            agreementBar
        );
    }

    if (
        readyPanel &&
        agreementStatus.totalAgreed >= 2
    ) {
        chatBody.appendChild(
            readyPanel
        );
    }

    if (!messages.length) {
        const empty =
            document.createElement("div");

        empty.className =
            "conversation-start";

        empty.innerHTML = `
            <i class="fas fa-comments"></i>
            <h3>Start a conversation</h3>
            <p>Say hello and discuss the item.</p>
        `;

        chatBody.appendChild(empty);

        return;
    }

    messages.forEach(message => {
        const element =
            document.createElement("div");

        const senderId =
            Number(message.sender_id);

        const own =
            senderId ===
            Number(window.currentUserId);

        element.className =
            `msg ${own ? "sent" : "received"}`;

        const messageType =
            message.message_type ||
            "text";

        if (
            messageType === "image" &&
            message.image_url
        ) {
            element.innerHTML = `
                <div class="message-image-wrapper">
                    <img
                        src="${escapeAttribute(
                            message.image_url
                        )}"
                        alt="Shared item"
                        class="message-image"
                        loading="lazy"
                    >
                </div>

                <small class="message-time">
                    ${formatMessageTime(
                        message.created_at
                    )}
                </small>
            `;
        } else {
            element.innerHTML = `
                <span>
                    ${escapeHtml(
                        message.message || ""
                    )}
                </span>

                <small class="message-time">
                    ${formatMessageTime(
                        message.created_at
                    )}
                </small>
            `;
        }

        chatBody.appendChild(
            element
        );
    });

    if (
        !preserveScroll ||
        wasNearBottom
    ) {
        scrollChatToBottom();
    } else {
        chatBody.scrollTop =
            previousTop +
            (
                chatBody.scrollHeight -
                previousHeight
            );
    }
}


/* =========================================================
   MESSAGE POLLING
   ========================================================= */

function startMessagePolling() {
    stopMessagePolling();

    if (!currentConversationId) {
        return;
    }

    messagePollingInterval =
        setInterval(async () => {
            if (
                document.hidden ||
                !currentConversationId
            ) {
                return;
            }

            const id =
                currentConversationId;

            await loadMessages(
                id,
                false
            );

            await loadConversations({
                silent: true
            });

        }, MESSAGE_POLL_INTERVAL);
}


function stopMessagePolling() {
    if (
        messagePollingInterval !== null
    ) {
        clearInterval(
            messagePollingInterval
        );

        messagePollingInterval =
            null;
    }
}

async function loadAgreementStatus() {
    if (
        !currentConversationId ||
        agreementPollRunning
    ) {
        return;
    }

    agreementPollRunning = true;

    try {
        const id =
            currentConversationId;

        const response = await fetch(
            `/api/conversations/${encodeURIComponent(
                id
            )}/agreement`,
            {
                method: "GET",
                headers: {
                    Accept:
                        "application/json"
                },
                cache: "no-store"
            }
        );

        if (response.status === 401) {
            window.location.href =
                "index.html";
            return;
        }

        if (response.status === 404) {
            return;
        }

        const data =
            await response.json();

        if (!response.ok || !data.success) {
            return;
        }

        const previousStatus =
            agreementStatus.status;

        agreementStatus = {
            userAgreed: Boolean(
                data.userAgreed ??
                data.user_agreed ??
                false
            ),

            otherUserAgreed: Boolean(
                data.otherUserAgreed ??
                data.other_user_agreed ??
                false
            ),

            totalAgreed: Number(
                data.totalAgreed ??
                data.total_agreed ??
                0
            ),

            status:
                data.status ||
                "pending",

            barangay:
                data.barangay ||
                agreementStatus.barangay ||
                null
        };

        if (
            agreementStatus.totalAgreed >= 2 ||
            agreementStatus.status ===
                "both_agreed"
        ) {
            agreementStatus.totalAgreed = 2;
            agreementStatus.status =
                "both_agreed";

            agreementStatus.otherUserAgreed =
                true;

            updateAgreementUI();

            showTransactionReadyPanel();

            if (
                previousStatus !==
                    "both_agreed"
            ) {
                showToast(
                    "Both parties agreed. The transaction is now ready for admin review."
                );
            }

        } else {
            updateAgreementUI();
        }

    } catch (error) {
        console.error(
            "loadAgreementStatus:",
            error
        );
    } finally {
        agreementPollRunning = false;
    }
}


function updateAgreementUI() {
    const count =
        document.getElementById(
            "agreementCount"
        );

    const button =
        document.getElementById(
            "agreeDealBtn"
        );

    const description =
        document.getElementById(
            "agreementDescription"
        );

    const total = Math.max(
        0,
        Math.min(
            2,
            Number(
                agreementStatus.totalAgreed ||
                0
            )
        )
    );

    if (count) {
        count.textContent =
            `Agreement: ${total}/2`;
    }

    if (
        agreementStatus.userAgreed
    ) {
        if (button) {
            button.disabled = true;
            button.textContent =
                "Agreed";
        }

        if (description) {
            if (
                agreementStatus.otherUserAgreed ||
                total >= 2
            ) {
                description.textContent =
                    "Both parties agreed. The transaction is ready for admin review.";
            } else {
                description.textContent =
                    "You agreed. Waiting for the other party.";
            }
        }

    } else {
        if (button) {
            button.disabled = false;
            button.textContent =
                "Agree";
        }

        if (description) {
            description.textContent =
                "Both parties must agree before the transaction can be reviewed by an admin.";
        }
    }
}

/* =========================================================
   CONFIRM AGREEMENT
   ========================================================= */

async function confirmAgreement() {
    if (!currentConversationId) {
        return;
    }

    if (agreementStatus.userAgreed) {
        return;
    }

    let barangayToSend = agreementStatus.barangay || "";
    if (!barangayToSend) {
        const barangayInput = document.getElementById("agreementBarangayInput");
        barangayToSend = barangayInput ? barangayInput.value.trim() : "";

        if (!barangayToSend) {
            alert("Please enter a barangay hall meetup location before agreeing.");
            if (barangayInput) barangayInput.focus();
            return;
        }
    }

    const button =
        document.getElementById(
            "confirmAgreementBtn"
        );

    if (button) {
        button.disabled = true;

        button.innerHTML = `
            <i class="fas fa-spinner fa-spin"></i>
            Confirming...
        `;
    }

    try {
        const response = await fetch(
            `/api/conversations/${encodeURIComponent(
                currentConversationId
            )}/agreement`,
            {
                method: "POST",
                headers: {
                    "Content-Type":
                        "application/json",
                    Accept:
                        "application/json"
                },
                body: JSON.stringify({
                    agreed: true,
                    barangay: barangayToSend
                })
            }
        );

        const data =
            await response.json();

        console.log(
            "Agreement response:",
            data
        );

        if (!response.ok || !data.success) {
            throw new Error(
                data.message ||
                "Unable to save your agreement."
            );
        }

        /*
         * Update using backend response.
         */

        agreementStatus.userAgreed =
            Boolean(
                data.userAgreed ??
                data.user_agreed ??
                true
            );

        agreementStatus.otherUserAgreed =
            Boolean(
                data.otherUserAgreed ??

async function getItemDetails() {

    const conversation =
        currentConversation || {};

    /*
     * First try data already returned
     * by conversation endpoint.
     */

    const hasDetails =
        conversation.item_description ||
        conversation.description ||
        conversation.item_category ||
        conversation.category ||
        conversation.item_location ||
        conversation.location ||
        conversation.item_image ||
        conversation.image_url;

    if (hasDetails) {
        itemDetailsCache =
            conversation;

        return itemDetailsCache;
    }

    const itemId =
        conversation.item_id ||
        conversation.itemId;

    if (!itemId) {
        itemDetailsCache =
            conversation;

        return itemDetailsCache;
    }

    const endpoints = [
        `/api/items/${itemId}`,
        `/api/items/${itemId}/details`,
        `/api/item/${itemId}`,
        `/api/item/${itemId}/details`
    ];

    for (const endpoint of endpoints) {
        try {
            const response =
                await fetch(
                    endpoint,
                    {
                        headers: {
                            Accept:
                                "application/json"
                        },
                        cache:
                            "no-store"
                    }
                );

            if (!response.ok) {
                continue;
            }

            const data =
                await response.json();

            if (!data.success) {
                continue;
            }

            itemDetailsCache =
                data.item ||
                data.data ||
                data;

            return itemDetailsCache;

        } catch (error) {
            console.warn(
                "Item endpoint failed:",
                endpoint,
                error
            );
        }
    }

    itemDetailsCache =
        conversation;

    return itemDetailsCache;
}

function firstValue(

    fallback = ""
) {
    if (!object) {
        return fallback;
    }

            value !== null &&
            String(value).trim() !== ""
        ) {
            return value;
        }
    }

    return fallback;
}

function handleLoginSubmit() {
    if (loginForm) {
        loginForm.addEventListener('submit', async (e) => {
            e.preventDefault();
            
            const isTrusted = localStorage.getItem('device_trusted') === 'true';
            const submitBtn = loginForm.querySelector('button');
            const originalText = submitBtn.innerText;
            const emailField = loginForm.querySelector('input[name="email"]') || loginForm.querySelector('input[type="email"]');
            const passwordField = loginForm.querySelector('input[name="password"]') || loginForm.querySelector('input[type="password"]');

            if (!emailField || !passwordField) return;

            const email = emailField.value;
            const password = passwordField.value;

            errorDiv.textContent = ""; 
            submitBtn.disabled = true;
            submitBtn.innerText = "Verifying...";
            submitBtn.style.opacity = "0.7";

            try {
                const response = await fetch('http://localhost:3000/login', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ email, password, isTrustedDevice: isTrusted })
                });

                // --- 1. HANDLE BRUTE FORCE LOGGING ---
                if (response.status === 429) {
                    const rateLimitData = await response.json();
                    
                    // Send to Audit Log
                    await fetch('http://localhost:3000/api/admin/audit_logs', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            action: 'SECURITY_BRUTE_FORCE_BLOCK',
                            guest_identifier: email,
                            details: JSON.stringify({ reason: "Rate limit exceeded", retryAfter: rateLimitData.retryAfter })
                        })
                    });

                    if (rateLimitData.retryAfter) startLoginTimer(rateLimitData.retryAfter);
                    throw new Error(rateLimitData.message || "Too many attempts.");
                }

                const result = await response.json();

                // --- 2. CHECK PARA SA SUSPENDED ACCOUNT ---
                if (response.status === 403 || result.isSuspended || (result.message && result.message.toLowerCase().includes('suspend'))) {
                    showSuspensionModal(result); // ðŸŸ¢ PINALITAN LANG ITO: ipinasa ang buong result object
                    return;
                }

                // --- 3. HANDLE INVALID CREDENTIALS LOGGING ---
                if (!response.ok) {
                    await fetch('http://localhost:3000/api/admin/audit_logs', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            action: 'SECURITY_LOGIN_FAILURE',
                            guest_identifier: email,
                            details: JSON.stringify({ reason: "Invalid Credentials", attempt_time: new Date().toISOString() })
                        })
                    });
                    
                    throw new Error(result.message || "Login failed.");
                }

                if (result.success === false) {
                    throw new Error(result.message || "Login failed.");
                }

                // --- 4. PROCEED TO MFA OR DASHBOARD ---
                if (result.mfaRequired) {
                    const overlay = document.getElementById('otp-overlay');
                    if (overlay) {
                        overlay.style.display = 'flex'; 
                        overlay.setAttribute('data-mode', 'login');
                        overlay.setAttribute('data-email', email); 
                    }
                } else if (result.success) {
                    window.location.href = '/dashboard.html';
                }

            } catch (err) {
                errorDiv.textContent = err.message;
                errorDiv.style.color = "red";
            } finally {
                submitBtn.disabled = false;
                submitBtn.innerText = originalText;
                submitBtn.style.opacity = "1";
            }
        });
    }
}

// Function para ipakita ang Suspension Pop-up Modal
function showSuspensionModal(data) {
    const modal = document.getElementById('suspension-modal');
    const reasonText = document.getElementById('suspension-reason-text');
    const durationText = document.getElementById('suspension-duration-text');
    
    // 1. Handling for Reason (Supports array or string, snake_case or camelCase)
    let rawReason = data.suspend_reasons || data.reason || data.suspensionReason;
    if (Array.isArray(rawReason)) {
        reasonText.innerText = rawReason.length > 0 ? rawReason.join(", ") : "Violation of ReFoundly community guidelines.";
    } else if (typeof rawReason === 'string' && rawReason.trim() !== '') {
        reasonText.innerText = rawReason;
    } else {
        reasonText.innerText = "Violation of ReFoundly community guidelines.";
    }

    // 2. Handling for Duration / Date (Supports suspended_until or suspendedUntil)
    const rawUntil = data.suspended_until || data.suspendedUntil;

    if (data.daysRemaining) {
        durationText.innerText = `${data.daysRemaining} day${data.daysRemaining > 1 ? 's' : ''} remaining`;
    } else if (rawUntil) {
        const untilDate = new Date(rawUntil);
        const now = new Date();
        const diffMs = untilDate - now;
        
        if (!isNaN(untilDate.getTime()) && diffMs > 0) {
            const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
            durationText.innerText = `${diffDays} day${diffDays > 1 ? 's' : ''} remaining (Until ${untilDate.toLocaleDateString()})`;
        } else if (!isNaN(untilDate.getTime())) {
            durationText.innerText = `Suspended until ${untilDate.toLocaleDateString()}`;
        } else {
            durationText.innerText = "Permanent / Indefinite Suspension";
        }
    } else {
        durationText.innerText = "Permanent / Indefinite Suspension";
    }

    if (modal) modal.style.display = 'flex';
}

function closeSuspensionModal() {
    const modal = document.getElementById('suspension-modal');
    if (modal) {
        modal.style.display = 'none';
    }
}
