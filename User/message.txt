document.addEventListener("DOMContentLoaded", init);

async function init() {
    try {
        createAgreementModal();
        createItemDetailsModal();

        setupSearch();
        setupSendMessage();
        setupAgreementEvents();
        setupItemDetailsEvents();

        await loadCurrentUser();

        if (!window.currentUserId) {
            console.warn("No logged-in user found.");
            showEmptyState("Please log in to use messages.");
            return;
        }

        await loadConversations();

        const params = new URLSearchParams(window.location.search);

        const targetUserId =
            params.get("userId") ||
            params.get("user") ||
            null;

        const itemId =
            params.get("itemId") ||
            params.get("item") ||
            null;

        if (targetUserId && itemId) {
            await startConversationFromItem(
                targetUserId,
                itemId
            );
        } else if (targetUserId) {
            await startConversationWithUser(
                targetUserId
            );
        } else {
            showEmptyState();
        }

        startConversationListPolling();

    } catch (error) {
        console.error("Message page initialization error:", error);
    }
}


/* =========================================================
   USER
   ========================================================= */

async function loadCurrentUser() {
    try {
        const response = await fetch(
            "/api/get-session-info",
            {
                method: "GET",
                headers: {
                    Accept: "application/json"
                },
                cache: "no-store"
            }
        );

        if (response.status === 401) {
            window.location.href = "index.html";
            return;
        }

        if (!response.ok) {
            throw new Error("Failed to get session.");
        }

        const data = await response.json();

        if (data.success && data.userId) {
            window.currentUserId = Number(data.userId);
        }

    } catch (error) {
        console.error("loadCurrentUser:", error);
    }
}


/* =========================================================
   CONVERSATIONS
   ========================================================= */

async function loadConversations(options = {}) {
    const silent = options.silent === true;

    if (conversationPollRunning) {
        return conversations;
    }

    conversationPollRunning = true;

    try {
        const response = await fetch(
            "/api/conversations",
            {
                method: "GET",
                headers: {
                    Accept: "application/json"
                },
                cache: "no-store"
            }
        );

        if (response.status === 401) {
            window.location.href = "index.html";
            return [];
        }

        const data = await response.json();

        if (!response.ok || !data.success) {
            throw new Error(
                data.message ||
                "Unable to load conversations."
            );
        }

        const newList =
            Array.isArray(data.conversations)
                ? data.conversations
                : [];

        const changed =
            conversationsChanged(
                conversations,
                newList
            );

        conversations = newList;

        if (!silent || changed) {
            renderConversationList(
                conversations
            );
        }

        return conversations;

    } catch (error) {
        console.error(
            "loadConversations:",
            error
        );

        if (!silent && chatList) {
            chatList.innerHTML = `
                <div class="empty-chat-list">
                    <i class="fas fa-exclamation-circle"></i>
                    <p>Unable to load conversations.</p>
                </div>
            `;
        }

        return [];
    } finally {
        conversationPollRunning = false;
    }
}


function conversationsChanged(oldList, newList) {
    if (oldList.length !== newList.length) {
        return true;
    }

    for (let i = 0; i < newList.length; i++) {
        const oldItem = oldList[i];
        const newItem = newList[i];

        if (
            String(oldItem?.conversation_id) !==
            String(newItem?.conversation_id)
        ) {
            return true;
        }

        if (
            String(oldItem?.last_message || "") !==
            String(newItem?.last_message || "")
        ) {
            return true;
        }

        if (
            Number(oldItem?.unread_count || 0) !==
            Number(newItem?.unread_count || 0)
        ) {
            return true;
        }
    }

    return false;
}


function renderConversationList(list) {
    if (!chatList) return;

    chatList.innerHTML = "";

    if (!list || list.length === 0) {
        chatList.innerHTML = `
            <div class="empty-chat-list">
                <i class="fas fa-comments"></i>
                <p>No conversations yet.</p>
            </div>
        `;
        return;
    }

    list.forEach(conversation => {
        const conversationId =
            conversation.conversation_id;

        const name =
            conversation.other_user_name ||
            conversation.other_user_email ||
            "Unknown User";

        const lastMessage =
            conversation.last_message ||
            "No messages yet";

        const unread =
            Number(conversation.unread_count || 0);

        const item =
            document.createElement("div");

        item.className = "chat-item";

        if (
            Number(conversationId) ===
            Number(currentConversationId)
        ) {
            item.classList.add("active");
        }

        const avatarName =
            encodeURIComponent(name);

        item.innerHTML = `
            <img
                src="https://ui-avatars.com/api/?name=${avatarName}&background=6B9E45&color=fff"
                alt="User"
                loading="lazy"
            >

            <div class="preview-text">
                <h4>${escapeHtml(name)}</h4>

                <p>
                    ${escapeHtml(lastMessage)}
                </p>

                ${
                    unread > 0
                        ? `<span class="chat-unread">${unread}</span>`
                        : ""
                }
            </div>
        `;

        item.addEventListener(
            "click",
            () => openConversation(conversationId)
        );

        chatList.appendChild(item);
    });
}


/* =========================================================
   START CONVERSATION
   ========================================================= */

async function startConversationFromItem(
    targetUserId,
    itemId
) {
    if (isSelf(targetUserId)) {
        showEmptyState();
        return;
    }

    try {
        const response = await fetch(
            "/api/conversations/start",
            {
                method: "POST",
                headers: {
                    "Content-Type":
                        "application/json",
                    Accept:
                        "application/json"
                },
                body: JSON.stringify({
                    userId:
                        Number(targetUserId),
                    itemId:
                        Number(itemId)
                })
            }
        );

        const data =
            await response.json();

        if (!response.ok || !data.success) {
            alert(
                data.message ||
                "Unable to start conversation."
            );
            return;
        }

        const conversationId =
            data.conversationId ||
            data.conversation_id;

        if (!conversationId) {
            alert(
                "Conversation was created but no conversation ID was returned."
            );
            return;
        }

        await loadConversations();

        await openConversation(
            conversationId
        );

        window.history.replaceState(
            {},
            document.title,
            "message.html"
        );

    } catch (error) {
        console.error(
            "startConversationFromItem:",
            error
        );

        alert(
            "Unable to start conversation."
        );
    }
}

/* =========================================================
   OPEN CONVERSATION
   ========================================================= */

async function openConversation(
    conversationId
) {
    if (!conversationId) return;

    stopMessagePolling();
    stopAgreementPolling();

    currentConversationId =
        Number(conversationId);

    currentConversation = null;
    currentMessages = [];

    itemDetailsCache = null;
    lastMessageSignature = "";

    resetAgreementStatus();

    updateConversationHeader();

    if (chatFooter) {
        chatFooter.style.display = "none";
    }

    if (chatBody) {
        chatBody.innerHTML = `
            <div class="conversation-loading">
                <i class="fas fa-spinner fa-spin"></i>
                <p>Loading conversation...</p>
            </div>
        `;
    }

    try {
        const response = await fetch(
            `/api/conversations/${encodeURIComponent(
                currentConversationId
            )}`,
            {
                method: "GET",
                headers: {
                    Accept:
                        "application/json"
                },
                cache: "no-store"
            }
        );

        if (response.status === 401) {
            window.location.href =
                "index.html";
            return;
        }

        const data =
            await response.json();

        if (!response.ok || !data.success) {
            throw new Error(
                data.message ||
                "Unable to open conversation."
            );
        }

        currentConversation =
            data.conversation ||
            data.data ||
            {};

        updateConversationHeader();

        if (chatFooter) {
            chatFooter.style.display =
                "flex";
        }

        renderAgreementBar();

        await loadMessages(
            currentConversationId,
            true
        );

        await loadAgreementStatus();

        renderConversationList(
            conversations
        );

        startMessagePolling();
        startAgreementPolling();

    } catch (error) {
        console.error(
            "openConversation:",
            error
        );

        if (chatBody) {
            chatBody.innerHTML = `
                <div class="empty-chat-list">
                    <i class="fas fa-exclamation-circle"></i>
                    <p>Unable to open conversation.</p>
                </div>
            `;
        }
    }
}


function updateConversationHeader() {
    if (!currentConversation) {
        if (currentUserName) {
            currentUserName.textContent =
                "Select a conversation";
        }

        if (currentItemName) {
            currentItemName.textContent = "";
        }

        renderItemFloatingCard();
        return;
    }

    if (currentUserName) {
        currentUserName.textContent =
            currentConversation.other_user_name ||
            currentConversation.other_user_email ||
            "User";
    }

    if (currentItemName) {
        currentItemName.textContent =
            currentConversation.item_name
                ? `Item: ${currentConversation.item_name}`
                : "";
    }

    renderItemFloatingCard();
}


/* =========================================================
   ITEM FLOATING CARD (maximize / minimize item info in-chat)

function renderItemFloatingCard() {
    const card = document.getElementById("itemFloatingCard");
    if (!card) return;

    if (!currentConversation) {
        card.innerHTML = "";
        card.style.display = "none";
        itemCardExpanded = false;
        return;
    }

    card.style.display = "block";

    const name = currentConversation.item_name || "Item";
    const type = currentConversation.report_type
        ? currentConversation.report_type.toUpperCase()
        : "";
    const image =
        currentConversation.image_path ||
        currentConversation.item_image ||
        currentConversation.image_url ||
        "";

    card.innerHTML = `
        <div class="item-floating-inner" id="itemFloatingToggle">
            ${
                image
                    ? `<img src="${escapeAttribute(image)}" class="item-floating-image" alt="Item photo">`
                    : `<div class="item-floating-image" style="display:flex;align-items:center;justify-content:center;"><i class="fas fa-box-open" style="color:#aaa;"></i></div>`
            }

            <div class="item-floating-info">
                <span class="item-floating-label">${type ? `${type} ITEM` : "REGARDING"}</span>
                <div class="item-floating-name">${escapeHtml(name)}</div>
                <div class="item-floating-subtitle">${itemCardExpanded ? "Tap to minimize" : "Tap to see full details"}</div>
            </div>

            <i class="fas ${itemCardExpanded ? "fa-chevron-up" : "fa-chevron-down"} item-floating-arrow"></i>
        </div>

        <div id="itemFloatingExpanded" class="item-floating-expanded" style="display:${itemCardExpanded ? "block" : "none"};">
            <div class="item-floating-loading">
                <i class="fas fa-spinner fa-spin"></i> Loading details...
            </div>
        </div>
    `;

    const toggle = document.getElementById("itemFloatingToggle");
    if (toggle) {
        toggle.addEventListener("click", toggleItemFloatingCard);
    }

    if (itemCardExpanded) {
        populateItemFloatingExpanded();
    }
}

async function toggleItemFloatingCard() {
    itemCardExpanded = !itemCardExpanded;
    renderItemFloatingCard();
}

async function populateItemFloatingExpanded() {
    const container = document.getElementById("itemFloatingExpanded");
    if (!container) return;

    try {
        const item = await getItemDetails();

        const category = firstValue(item, ["category", "item_category"], "N/A");
        const location = firstValue(item, ["location", "item_location"], "N/A");
        const date = firstValue(item, ["incident_date", "item_date", "created_at", "date_reported"], "");
        const time = firstValue(item, ["incident_time"], "");
        const description = firstValue(item, ["description", "item_description"], "No description provided.");

            <div class="item-floating-detail-row"><i class="fas fa-tag"></i> <span>${escapeHtml(category)}</span></div>

            <button type="button" class="item-floating-full-btn" id="itemFloatingFullBtn">

        console.error("populateItemFloatingExpanded:", error);
        container.innerHTML = `<div class="item-floating-detail-desc">Unable to load full item details right now.</div>`;
    }
}


/* =========================================================
   MESSAGES
   ========================================================= */

async function loadMessages(
    conversationId,
    initial = false
) {
    if (!conversationId) return [];

    if (messagePollRunning) {
        return currentMessages;
    }

    messagePollRunning = true;

    try {
        const response = await fetch(
            `/api/conversations/${encodeURIComponent(
                conversationId
            )}/messages`,
            {
                method: "GET",
                headers: {
                    Accept:
                        "application/json"
                },
                cache: "no-store"
            }
        );

        if (response.status === 401) {
            window.location.href =
                "index.html";
            return [];
        }

        const data =
            await response.json();

        if (!response.ok || !data.success) {
            throw new Error(
                data.message ||
                "Unable to load messages."
            );
        }

        const messages =
            Array.isArray(data.messages)
                ? data.messages
                : [];

        const signature =
            makeMessageSignature(
                messages
            );

        if (
            initial ||
            signature !== lastMessageSignature
        ) {
            renderMessages(
                messages,
                !initial
            );

            lastMessageSignature =
                signature;

            currentMessages =
                messages;
        }

        return messages;

    } catch (error) {
        console.error(
            "loadMessages:",
            error
        );

        if (initial && chatBody) {
            chatBody.innerHTML = `
                <div class="empty-chat-list">
                    <i class="fas fa-exclamation-circle"></i>
                    <p>Unable to load messages.</p>
                </div>
            `;
        }

        return [];
    } finally {
        messagePollRunning = false;
    }
}


function makeMessageSignature(
    messages
) {
    if (!messages.length) {
        return "empty";
    }

    return messages
        .map(message => {
            return [
                message.id ||
                    message.message_id ||
                    "",
                message.sender_id ||
                    "",
                message.message ||
                    "",
                message.message_type ||
                    "",
                message.image_url ||
                    "",
                message.created_at ||
                    ""
            ].join("|");
        })
        .join("||");
}


function renderMessages(
    messages,
    preserveScroll = true
) {
    if (!chatBody) return;

    const wasNearBottom =
        isChatNearBottom();

    const previousHeight =
        chatBody.scrollHeight;

    const previousTop =
        chatBody.scrollTop;

    const agreementBar =
        document.getElementById(
            "agreementBar"
        );

    const readyPanel =
        document.getElementById(
            "transactionReadyPanel"
        );

    chatBody.innerHTML = "";

    if (agreementBar) {
        chatBody.appendChild(
            agreementBar
        );
    }

    if (
        readyPanel &&
        agreementStatus.totalAgreed >= 2
    ) {
        chatBody.appendChild(
            readyPanel
        );
    }

    if (!messages.length) {
        const empty =
            document.createElement("div");

        empty.className =
            "conversation-start";

        empty.innerHTML = `
            <i class="fas fa-comments"></i>
            <h3>Start a conversation</h3>
            <p>Say hello and discuss the item.</p>
        `;

        chatBody.appendChild(empty);

        return;
    }

    messages.forEach(message => {
        const element =
            document.createElement("div");

        const senderId =
            Number(message.sender_id);

        const own =
            senderId ===
            Number(window.currentUserId);

        element.className =
            `msg ${own ? "sent" : "received"}`;

        const messageType =
            message.message_type ||
            "text";

        if (
            messageType === "image" &&
            message.image_url
        ) {
            element.innerHTML = `
                <div class="message-image-wrapper">
                    <img
                        src="${escapeAttribute(
                            message.image_url
                        )}"
                        alt="Shared item"
                        class="message-image"
                        loading="lazy"
                    >
                </div>

                <small class="message-time">
                    ${formatMessageTime(
                        message.created_at
                    )}
                </small>
            `;
        } else {
            element.innerHTML = `
                <span>
                    ${escapeHtml(
                        message.message || ""
                    )}
                </span>

                <small class="message-time">
                    ${formatMessageTime(
                        message.created_at
                    )}
                </small>
            `;
        }

        chatBody.appendChild(
            element
        );
    });

    if (
        !preserveScroll ||
        wasNearBottom
    ) {
        scrollChatToBottom();
    } else {
        chatBody.scrollTop =
            previousTop +
            (
                chatBody.scrollHeight -
                previousHeight
            );
    }
}


/* =========================================================
   MESSAGE POLLING
   ========================================================= */

function startMessagePolling() {
    stopMessagePolling();

    if (!currentConversationId) {
        return;
    }

    messagePollingInterval =
        setInterval(async () => {
            if (
                document.hidden ||
                !currentConversationId
            ) {
                return;
            }

            const id =
                currentConversationId;

            await loadMessages(
                id,
                false
            );

            await loadConversations({
                silent: true
            });

        }, MESSAGE_POLL_INTERVAL);
}


function stopMessagePolling() {
    if (
        messagePollingInterval !== null
    ) {
        clearInterval(
            messagePollingInterval
        );

        messagePollingInterval =
            null;
    }
}


/* =========================================================
   CONVERSATION POLLING
   ========================================================= */

function startConversationListPolling() {
    stopConversationListPolling();

    conversationPollingInterval =
        setInterval(async () => {
            if (document.hidden) {
                return;
            }

            await loadConversations({
                silent: true
            });

        }, CONVERSATION_POLL_INTERVAL);
}


function stopConversationListPolling() {
    if (
        conversationPollingInterval !== null
    ) {
        clearInterval(
            conversationPollingInterval
        );

        conversationPollingInterval =
            null;
    }
}


/* =========================================================
   AGREEMENT
   ========================================================= */

function resetAgreementStatus() {
    agreementStatus = {
        userAgreed: false,
        otherUserAgreed: false,
        totalAgreed: 0,
        status: "pending",
        barangay: null
    };
}

function hasUserConfirmedItem(messages = currentMessages) {
    if (!Array.isArray(messages) || !window.currentUserId) {
        return false;
    }

    return messages.some(message => {
        const senderId = Number(message.sender_id);

        const text = String(
            message.message || ""
        )
            .trim()
            .replace(/\s+/g, " ")
            .toLowerCase();

        return (
            senderId === Number(window.currentUserId) &&
            text === REQUIRED_CONFIRM_MESSAGE
        );
    });
}

        </div>

async function loadAgreementStatus() {
    if (
        !currentConversationId ||
        agreementPollRunning
    ) {
        return;
    }

    agreementPollRunning = true;

    try {
        const id =
            currentConversationId;

        const response = await fetch(
            `/api/conversations/${encodeURIComponent(
                id
            )}/agreement`,
            {
                method: "GET",
                headers: {
                    Accept:
                        "application/json"
                },
                cache: "no-store"
            }
        );

        if (response.status === 401) {
            window.location.href =
                "index.html";
            return;
        }

        if (response.status === 404) {
            return;
        }

        const data =
            await response.json();

        if (!response.ok || !data.success) {
            return;
        }

        const previousStatus =
            agreementStatus.status;

        agreementStatus = {
            userAgreed: Boolean(
                data.userAgreed ??
                data.user_agreed ??
                false
            ),

            otherUserAgreed: Boolean(
                data.otherUserAgreed ??
                data.other_user_agreed ??
                false
            ),

            totalAgreed: Number(
                data.totalAgreed ??
                data.total_agreed ??
                0
            ),

            status:
                data.status ||
                "pending",

            barangay:
                data.barangay ||
                agreementStatus.barangay ||
                null
        };

        if (
            agreementStatus.totalAgreed >= 2 ||
            agreementStatus.status ===
                "both_agreed"
        ) {
            agreementStatus.totalAgreed = 2;
            agreementStatus.status =
                "both_agreed";

            agreementStatus.otherUserAgreed =
                true;

            updateAgreementUI();

            showTransactionReadyPanel();

            if (
                previousStatus !==
                    "both_agreed"
            ) {
                showToast(
                    "Both parties agreed. The transaction is now ready for admin review."
                );
            }

        } else {
            updateAgreementUI();
        }

    } catch (error) {
        console.error(
            "loadAgreementStatus:",
            error
        );
    } finally {
        agreementPollRunning = false;
    }
}


function updateAgreementUI() {
    const count =
        document.getElementById(
            "agreementCount"
        );

    const button =
        document.getElementById(
            "agreeDealBtn"
        );

    const description =
        document.getElementById(
            "agreementDescription"
        );

    const total = Math.max(
        0,
        Math.min(
            2,
            Number(
                agreementStatus.totalAgreed ||
                0
            )
        )
    );

    if (count) {
        count.textContent =
            `Agreement: ${total}/2`;
    }

    if (
        agreementStatus.userAgreed
    ) {
        if (button) {
            button.disabled = true;
            button.textContent =
                "Agreed";
        }

        if (description) {
            if (
                agreementStatus.otherUserAgreed ||
                total >= 2
            ) {
                description.textContent =
                    "Both parties agreed. The transaction is ready for admin review.";
            } else {
                description.textContent =
                    "You agreed. Waiting for the other party.";
            }
        }

    } else {
        if (button) {
            button.disabled = false;
            button.textContent =
                "Agree";
        }

        if (description) {
            description.textContent =
                "Both parties must agree before the transaction can be reviewed by an admin.";
        }
    }
}

async function confirmAgreement() {
    if (!currentConversationId) {
        return;
    }

    if (agreementStatus.userAgreed) {
        return;
    }

    let barangayToSend = agreementStatus.barangay || "";
    if (!barangayToSend) {
        const barangayInput = document.getElementById("agreementBarangayInput");
        barangayToSend = barangayInput ? barangayInput.value.trim() : "";

        if (!barangayToSend) {
            alert("Please select a barangay hall meetup location before agreeing.");
            if (barangayInput) barangayInput.focus();
            return;
        }
    }

    const button =
        document.getElementById(
            "confirmAgreementBtn"
        );

    if (button) {
        button.disabled = true;

        button.innerHTML = `
            <i class="fas fa-spinner fa-spin"></i>
            Confirming...
        `;
    }

    try {
        const response = await fetch(
            `/api/conversations/${encodeURIComponent(
                currentConversationId
            )}/agreement`,
            {
                method: "POST",
                headers: {
                    "Content-Type":
                        "application/json",
                    Accept:
                        "application/json"
                },
                body: JSON.stringify({
                    agreed: true,
                    barangay: barangayToSend
                })
            }
        );

        const data =
            await response.json();

        console.log(
            "Agreement response:",
            data
        );

        if (!response.ok || !data.success) {
            throw new Error(
                data.message ||
                "Unable to save your agreement."
            );
        }

        agreementStatus.userAgreed =
            Boolean(
                data.userAgreed ??
                data.user_agreed ??
                true
            );

        agreementStatus.otherUserAgreed =
            Boolean(
                data.otherUserAgreed ??
                data.other_user_agreed ??
                false
            );

        agreementStatus.totalAgreed =
            Number(
                data.totalAgreed ??
                data.total_agreed ??
                (
                    agreementStatus.userAgreed
                        ? agreementStatus.totalAgreed + 1
                        : agreementStatus.totalAgreed
                )
            );

        agreementStatus.status =
            data.status ||
            (
                agreementStatus.totalAgreed >= 2
                    ? "both_agreed"
                    : "pending"
            );

        agreementStatus.barangay =
            data.barangay ||
            agreementStatus.barangay ||
            barangayToSend;

        if (
            agreementStatus.totalAgreed >= 2 ||
            agreementStatus.status ===
                "both_agreed"
        ) {
            agreementStatus.totalAgreed = 2;
            agreementStatus.status =
                "both_agreed";
            agreementStatus.otherUserAgreed =
                true;

            closeAgreementModal();

            updateAgreementUI();

            showTransactionReadyPanel();

            showToast(
                "Both parties agreed. The transaction is now ready for admin review."
            );

            await notifyAdminAgreementReady();

        } else {
            closeAgreementModal();

            updateAgreementUI();

            showToast(
                "You agreed to the deal. Waiting for the other party."
            );
        }

    } catch (error) {
        console.error(
            "confirmAgreement:",
            error
        );

        alert(
            error.message ||
            "Unable to save your agreement."
        );

    } finally {
        if (button) {
            button.disabled = false;

            button.innerHTML = `
                <i class="fas fa-check"></i>
                Yes, I Agree
            `;
        }
    }
}

async function sendMessage() {
    if (!currentConversationId) {
        alert(
            "Please select or start a conversation first."
        );
        return;
    }

    if (!messageInput) {
        return;
    }

    const message =
        messageInput.value.trim();

    if (!message) {
        return;
    }

    if (sendBtn) {
        sendBtn.disabled = true;
    }

    const original =
        messageInput.value;

    try {
        const response =
            await fetch(
                `/api/conversations/${encodeURIComponent(
                    currentConversationId
                )}/messages`,
                {
                    method: "POST",
                    headers: {
                        "Content-Type":
                            "application/json",
                        Accept:
                            "application/json"
                    },
                    body: JSON.stringify({
                        message
                    })
                }
            );

        const data =
            await response.json();

        if (!response.ok || !data.success) {
            throw new Error(
                data.message ||
                "Failed to send message."
            );
        }

        messageInput.value = "";

        lastMessageSignature = "";

        await loadMessages(
            currentConversationId,
            false
        );

        await loadConversations({
            silent: true
        });

    } catch (error) {
        console.error(
            "sendMessage:",
            error
        );

        messageInput.value =
            original;

        alert(
            error.message ||
            "Failed to send message."
        );

    } finally {
        if (sendBtn) {
            sendBtn.disabled = false;
        }

        messageInput.focus();
    }
}

                renderConversationList(

                );
                return;

            const filtered =
                conversations.filter(
                    conversation => {

                        const name =
                            String(
                                conversation.other_user_name ||
                                ""
                            ).toLowerCase();

                        const email =
                            String(
                                conversation.other_user_email ||
                                ""
                            ).toLowerCase();

                        const item =
                            String(
                                conversation.item_name ||
                                ""
                            ).toLowerCase();

                        return (
                            name.includes(search) ||
                            email.includes(search) ||
                            item.includes(search)
                        );
                    }
                );

            renderConversationList(
                filtered
            );
        }
    );
}

async function getItemDetails() {
    if (itemDetailsCache) {
        return itemDetailsCache;
    }

    const conversation =
        currentConversation || {};

    const itemId =
        conversation.item_id ||
        conversation.itemId;

    if (itemId) {
        const endpoints = [
            `/api/items/${itemId}`,
            `/api/items/${itemId}/details`,
            `/api/item/${itemId}`,
            `/api/item/${itemId}/details`
        ];

        for (const endpoint of endpoints) {
            try {
                const response =
                    await fetch(
                        endpoint,
                        {
                            headers: {
                                Accept:
                                    "application/json"
                            },
                            cache:
                                "no-store"
                        }
                    );

                if (!response.ok) {
                    continue;
                }

                const data =
                    await response.json();

                const itemObj =
                    data.item ||
                    data.data ||
                    (data.id || data.item_name || data.name ? data : null);

                if (itemObj) {
                    itemDetailsCache = {
                        ...conversation,
                        ...itemObj
                    };
                    return itemDetailsCache;
                }

            } catch (error) {
                console.warn(
                    "Item endpoint failed:",
                    endpoint,
                    error
                );
            }
        }
    }

    itemDetailsCache = conversation;
    return itemDetailsCache;
}

        </div>

}


function firstValue(
    object,
    keys,
    fallback = ""
) {
    if (!object) {
        return fallback;
    }

    for (const key of keys) {
        const value =
            object[key];

        if (
            value !== undefined &&
            value !== null &&
            String(value).trim() !== ""
        ) {
            return value;
        }
    }

    return fallback;
}


function formatDetailDate(
    value
) {
    if (!value) return "";

    const date =
        new Date(value);

    if (
        Number.isNaN(
            date.getTime()
        )
    ) {
        return String(value);
    }

    return date.toLocaleDateString(
        [],
        {
            month: "long",
            day: "numeric",
            year: "numeric"
        }
    );
}

/* =========================================================
   HELPERS
   ========================================================= */

function escapeHtml(value) {
    if (
        value === null ||
        value === undefined
    ) {
        return "";
    }

    return String(value)
        .replace(
            /&/g,
            "&amp;"
        )
        .replace(
            /</g,
            "&lt;"
        )
        .replace(
            />/g,
            "&gt;"
        )
        .replace(
            /"/g,
            "&quot;"
        )
        .replace(
            /'/g,
            "&#039;"
        );
}


function escapeAttribute(value) {
    return escapeHtml(value);
}


function formatMessageTime(
    value
) {
    if (!value) {
        return "";
    }

    const date =
        new Date(value);

    if (
        Number.isNaN(
            date.getTime()
        )
    ) {
        return "";
    }

    return date.toLocaleString(
        [],
        {
            month: "short",
            day: "numeric",
            hour: "numeric",
            minute: "2-digit"
        }
    );
}


function isChatNearBottom() {
    if (!chatBody) {
        return true;
    }

    const distance =
        chatBody.scrollHeight -
        chatBody.scrollTop -
        chatBody.clientHeight;

    return distance <= 100;
}


function scrollChatToBottom() {
    if (!chatBody) return;

    requestAnimationFrame(() => {
        chatBody.scrollTop =
            chatBody.scrollHeight;
    });
}
