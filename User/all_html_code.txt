async function setupChatButton(item) {

    const button =
        document.getElementById("chatUserBtn");

    const message =
        document.getElementById("chatUserMessage");


    if (!button) {
        return;
    }


    try {

        const response =
            await fetch(
                "/api/get-session-info",
                {
                    headers: {
                        "Accept":
                            "application/json"
                    }
                }
            );


        const data =
            await response.json();


        if (!data.success) {

            button.disabled = true;

            button.innerHTML = `
                <i class="fas fa-lock"></i>
                Login Required
            `;

            return;
        }


        const currentUserId =
            Number(data.userId);


        const itemOwnerId =
            Number(item.user_id);


        /*

            If the current user owns this item,
            they cannot chat with themselves.
        */

        if (
            currentUserId ===
            itemOwnerId
        ) {

            button.disabled = true;

            button.innerHTML = `
                <i class="fas fa-user"></i>
                Your Post
            `;

            message.textContent =
                "You cannot chat with yourself about your own post.";

            return;
        }


        /*
            Item must be published.
        */

        if (
            item.status !==
            "Published"
        ) {

            button.disabled = true;

            button.innerHTML = `
                <i class="fas fa-lock"></i>
                Chat Unavailable
            `;

            message.textContent =
                "Chat is unavailable for this item.";

            return;

            Everything is okay.

        button.disabled = false;

        button.innerHTML = `
            <i class="fas fa-comments"></i>
            Chat with User
