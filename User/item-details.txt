async function setupChatButton(item) {

    const button =
        document.getElementById("chatUserBtn");

    const message =
        document.getElementById("chatUserMessage");


    if (!button) {
        return;
    }


    try {

        const response =
            await fetch(
                "/api/get-session-info",
                {
                    headers: {
                        "Accept":
                            "application/json"
                    }
                }
            );


        const data =
            await response.json();


        if (!data.success) {

            button.disabled = true;

            button.innerHTML = `
                <i class="fas fa-lock"></i>
                Login Required
            `;

            return;
        }


        const currentUserId =
            Number(data.userId);


        const itemOwnerId =
            Number(item.user_id);


        /*
            IMPORTANT:

            If the current user owns this item,
            they cannot chat with themselves.
        */

        if (
            currentUserId ===
            itemOwnerId
        ) {

            button.disabled = true;

            button.innerHTML = `
                <i class="fas fa-user"></i>
                Your Post
            `;

            message.textContent =
                "You cannot chat with yourself about your own post.";

            return;
        }


        /*
            Item must be published.
        */

        if (
            item.status !==
            "Published"
        ) {

            button.disabled = true;

            button.innerHTML = `
                <i class="fas fa-lock"></i>
                Chat Unavailable
            `;

            message.textContent =
                "Chat is unavailable for this item.";

            return;
        }


        /*
            Everything is okay.
        */

        button.disabled = false;

        button.innerHTML = `
            <i class="fas fa-comments"></i>
            Chat with User
        `;


        button.onclick =
            async () => {

                button.disabled = true;

                button.innerHTML = `
                    <i class="fas fa-spinner fa-spin"></i>
                    Opening Chat...
                `;


                try {

                    const response =
                        await fetch(
                            "/api/conversations/start",
                            {
                                method: "POST",

                                headers: {
                                    "Content-Type":
                                        "application/json",

                                    "Accept":
                                        "application/json"
                                },

                                body:
                                    JSON.stringify({
                                        itemId:
                                            item.id
                                    })
                            }
                        );


                    const result =
                        await response.json();


                    if (
                        !response.ok ||
                        !result.success
                    ) {

                        alert(
                            result.message ||
                            "Unable to start chat."
                        );

                        button.disabled = false;

                        button.innerHTML = `
                            <i class="fas fa-comments"></i>
                            Chat with User
                        `;

                        return;
                    }


                    /*
                        Conversation created
                        or existing conversation found.
                    */

                    window.location.href =
                        `message.html?itemId=${encodeURIComponent(item.id)}`;


                } catch (error) {

                    console.error(
                        "Chat start error:",
                        error
                    );

                    alert(
                        "Unable to connect to messaging."
                    );

                    button.disabled = false;

                    button.innerHTML = `
                        <i class="fas fa-comments"></i>
                        Chat with User
                    `;
                }
            };


    } catch (error) {

        console.error(
            "Chat button setup error:",
            error
        );

        button.disabled = true;

        button.innerHTML = `
            <i class="fas fa-exclamation-circle"></i>
            Chat Unavailable
        `;
    }
}
